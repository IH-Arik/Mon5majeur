create a private league api

curl --location 'https://api.mon5majeur.com/api/private-leagues/' \
--header 'Content-Type: application/json' \
--header 'Accept: application/json' \
--header 'Authorization: Bearer eyJhbGciOiJIUz' \
--data '{
  "leauge_name": "Sadid Leauge",
  "leauge_logo": "atlanta_hawks",
  "leauge_description": "Sadid",
  "team_budget": "100M",
  "max_team_number": "10"

kick team from private league api

--header 'Content-Type: application/json' \
--header 'Accept: application/json' \
--header 'Authorization: Bearer eyJhbGciOiJIUz' \
--data '{
  "team_id": 3,
  "league_id": 10
}'

{
